import gi
gi.require_version('Gst', '1.0')
from gi.repository import Gst, GLib
import os
import sys

# --- DOSYA YOLLARI ---
CURRENT_DIR = os.path.dirname(os.path.abspath(__file__))
HEF_PATH = os.path.join(CURRENT_DIR, "yolov8s-pose.hef")
JSON_CONFIG_PATH = os.path.join(CURRENT_DIR, "yolov8_pose_config.json")

# Post-process (.so) dosyasını bulmaya çalış
possible_so_paths = [
    os.path.join(CURRENT_DIR, "resources/libyolo_hailortpp_post.so"),
    os.path.join(CURRENT_DIR, "libyolo_hailortpp_post.so"),
    "/usr/lib/aarch64-linux-gnu/hailo/tappas/post_processes/libyolo_hailortpp_post.so"
]

POST_PROC_PATH = None
for path in possible_so_paths:
    if os.path.exists(path):
        POST_PROC_PATH = path
        break

# --- KONTROLLER ---
if not os.path.exists(HEF_PATH):
    print(f"HATA: Model dosyası bulunamadı: {HEF_PATH}")
    sys.exit(1)
    
config_param = ""
if os.path.exists(JSON_CONFIG_PATH):
    print(f"Config bulundu: {JSON_CONFIG_PATH}")
    config_param = f"config-path={JSON_CONFIG_PATH}"

if POST_PROC_PATH is None:
    print("HATA: Post-process (.so) kütüphanesi bulunamadı!")
    sys.exit(1)

def create_pipeline():
    # GÜNCELLENMİŞ PIPELINE:
    # 1. Kaynak (Kamera) -> 640x480
    # 2. Videoscale -> 640x640 (AI için)
    # 3. Hailonet (AI İşlemi)
    # 4. Hailofilter (Sonuçları Düzenleme)
    # 5. HAILOOVERLAY (ÇİZİM YAPAN PARÇA BURASI!)
    # 6. Ekrana Basma
    
    pipeline_str = f"""
        libcamerasrc ! video/x-raw, width=640, height=480, format=RGB !
        videoscale ! video/x-raw, width=640, height=640 !
        queue !
        hailonet hef-path={HEF_PATH} !
        queue !
        hailofilter so-path={POST_PROC_PATH} {config_param} qos=false !
        queue !
        hailooverlay !
        queue !
        videoconvert !
        fpsdisplaysink video-sink=autovideosink text-overlay=false sync=false
    """
    
    print("Pipeline başlatılıyor...")
    try:
        pipeline = Gst.parse_launch(pipeline_str)
    except Exception as e:
        print(f"Pipeline oluşturma hatası: {e}")
        sys.exit(1)
    return pipeline

def bus_call(bus, message, loop):
    t = message.type
    if t == Gst.MessageType.EOS:
        print("Yayın bitti.")
        loop.quit()
    elif t == Gst.MessageType.ERROR:
        err, debug = message.parse_error()
        print(f"Hata: {err}, {debug}")
        loop.quit()
    return True

if __name__ == "__main__":
    Gst.init(None)
    pipeline = create_pipeline()
    loop = GLib.MainLoop()
    
    bus = pipeline.get_bus()
    bus.add_signal_watch()
    bus.connect("message", bus_call, loop)

    pipeline.set_state(Gst.State.PLAYING)
    try:
        print("ÇALIŞIYOR: Çıkmak için Ctrl+C yapın.")
        loop.run()
    except KeyboardInterrupt:
        print("Kapatılıyor...")
        pipeline.set_state(Gst.State.NULL)
